Задача 5: Получение последнего элемента массива

function getLastElement(array) { if (array.length === 0) return undefined; return array[array.length - 1]; }

Задача 6: Проверка чётных чисел (forEach)

function logEven(array) { array.forEach(function(element) { if (typeof element === 'number' && element % 2 === 0) { console.log(element); } }); }

Задача 7: Удвоение элементов массива (map)

function doubleElements(array) { return array.map(function(element) { return element * 2; }); }
